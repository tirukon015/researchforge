# ---------- Web framework ----------
fastapi                 # the API framework
uvicorn[standard]       # the server that runs FastAPI
python-multipart        # required for file uploads


# ---------- Configuration ----------
pydantic                # data validation for API requests/responses
pydantic-settings       # loads and validates settings from env vars
python-dotenv           # reads the .env file during local development


# ---------- Database ----------
psycopg[binary]         # PostgreSQL driver
sqlalchemy              # database toolkit / query builder
supabase                # Supabase client (storage + database helpers)
pgvector                # Python support for the pgvector column type


# ---------- Testing ----------
pytest
pytest-asyncio          # for testing async FastAPI endpoints
httpx                   # test client for calling the API in tests


# ---------- Code quality ----------
ruff                    # linter + formatter (fast, replaces several tools)
mypy                    # static type checking
